## 一、集成测试概述

### 1.1 定义
**集成测试**验证多个模块协作的正确性：
- API 路由 + 业务服务 + 数据库
- Agent + LLM + 数据库
- Celery 任务 + Redis + 数据库

### 1.2 与单元测试的区别

| 维度 | 单元测试 | 集成测试 |
|------|---------|---------|
| **范围** | 单个函数/类 | 多个模块协作 |
| **依赖** | Mock 外部依赖 | 使用真实依赖 |
| **速度** | 快（毫秒级） | 慢（秒级） |
| **数据库** | Mock | 真实测试数据库 |
| **比例** | 70% | 25% |

# tests/integration/test_api/test_data_sources.py
import pytest
from fastapi import status

class TestDataSourceAPI:
    """数据源 API 集成测试"""
    
    def test_create_data_source_saves_to_database(
        self, client, auth_headers, db_session
    ):
        """创建数据源应保存到数据库"""
        # Arrange
        payload = {
            "name": "淘宝店A",
            "platform": "taobao",
            "config": {
                "api_key": "test_key",
                "shop_id": "12345"
            }
        }
        
        # Act
        response = client.post(
            "/api/v1/data_sources",
            json=payload,
            headers=auth_headers
        )
        
        # Assert
        assert response.status_code == status.HTTP_201_CREATED
        data = response.json()
        assert data["name"] == "淘宝店A"
        assert data["platform"] == "taobao"
        
        # 验证数据库
        from app.models.data_source import DataSource
        ds = db_session.query(DataSource).filter_by(
            name="淘宝店A"
        ).first()
        assert ds is not None
        assert ds.platform == "taobao"
    
    def test_list_data_sources_with_pagination(
        self, client, auth_headers, db_session
    ):
        """列表数据源应支持分页"""
        # Arrange - 创建 15 个数据源
        from app.models.data_source import DataSource
        from tests.factories import DataSourceFactory
        
        for i in range(15):
            ds = DataSourceFactory.create(name=f"店铺{i}")
            db_session.add(ds)
        db_session.commit()
        
        # Act - 第 1 页
        response = client.get(
            "/api/v1/data_sources?page=1&page_size=10",
            headers=auth_headers
        )
        
        # Assert
        assert response.status_code == status.HTTP_200_OK
        data = response.json()
        assert len(data["items"]) == 10
        assert data["total"] == 15
        assert data["page"] == 1
        assert data["pages"] == 2
    
    def test_delete_data_source_cascades_to_raw_data(
        self, client, auth_headers, db_session
    ):
        """删除数据源应级联删除原始数据"""
        # Arrange
        from app.models.data_source import DataSource
        from app.models.raw_data import RawData
        
        ds = DataSource(name="测试店铺", platform="taobao")
        db_session.add(ds)
        db_session.commit()
        
        # 添加原始数据
        raw = RawData(data_source_id=ds.id, content={"test": "data"})
        db_session.add(raw)
        db_session.commit()
        
        # Act - 删除数据源
        response = client.delete(
            f"/api/v1/data_sources/{ds.id}",
            headers=auth_headers
        )
        
        # Assert
        assert response.status_code == status.HTTP_204_NO_CONTENT
        
        # 验证级联删除
        assert db_session.query(DataSource).filter_by(id=ds.id).first() is None
        assert db_session.query(RawData).filter_by(data_source_id=ds.id).first() is None

import pytest
from app.agents.cleaning_agent import DataCleaningAgent
from app.agents.evaluation_agent import QualityEvaluationAgent
from app.agents.extraction_agent import KnowledgeExtractionAgent

class TestAgentChain:
    """Agent 链式调用集成测试"""
    
    def test_cleaning_to_evaluation_pipeline(self, db_session):
        """清洗 → 评估流水线"""
        # Arrange
        cleaning_agent = DataCleaningAgent()
        evaluation_agent = QualityEvaluationAgent()
        
        raw_text = "用户: 如何申请退款？\nAI: 在订单详情页点击申请退款。"
        
        # Act - 清洗
        cleaned = cleaning_agent.clean({"raw_text": raw_text})
        
        # Act - 评估
        evaluation = evaluation_agent.evaluate(cleaned)
        
        # Assert
        assert cleaned["user_message"] == "如何申请退款？"
        assert evaluation["overall_score"] >= 0
        assert evaluation["overall_score"] <= 100
    
    def test_full_pipeline_cleaning_to_knowledge(self, db_session):
        """完整流水线：清洗 → 评估 → 知识提取"""
        # Arrange
        cleaning_agent = DataCleaningAgent()
        evaluation_agent = QualityEvaluationAgent()
        extraction_agent = KnowledgeExtractionAgent()
        
        raw_text = "用户: 如何申请退款？\nAI: 在订单详情页点击申请退款。"
        
        # Act
        cleaned = cleaning_agent.clean({"raw_text": raw_text})
        evaluation = evaluation_agent.evaluate(cleaned)
        
        # 仅当质量合格时提取知识
        if evaluation["overall_score"] >= 70:
            knowledge = extraction_agent.extract_faq(cleaned)
        
        # Assert
        assert knowledge["question"] == "如何申请退款？"
        assert "申请退款" in knowledge["answer"]
        assert len(knowledge["tags"]) > 0

## 八、事务和并发测试

### 8.1 事务隔离测试

```python
def test_concurrent_updates_maintain_data_integrity(client, auth_headers, db_session):
    """并发更新应保持数据完整性"""
    import concurrent.futures
    from app.models.data_source import DataSource
    
    # Arrange
    ds = DataSource(name="测试店铺", platform="taobao", sync_count=0)
    db_session.add(ds)
    db_session.commit()
    
    def increment_sync_count():
        response = client.patch(
            f"/api/v1/data_sources/{ds.id}/increment_sync",
            headers=auth_headers
        )
        return response.status_code
    
    # Act - 10 个并发请求
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(increment_sync_count) for _ in range(10)]
        results = [f.result() for f in futures]
    
    # Assert
    assert all(r == 200 for r in results)
    
    # 验证最终计数
    db_session.refresh(ds)
    assert ds.sync_count == 10
```
