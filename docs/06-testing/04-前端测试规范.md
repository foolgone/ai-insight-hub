## 四、组件单元测试

### 4.1 Button 组件测试

```tsx
// src/components/Button/Button.test.tsx
import { render, screen, fireEvent } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { Button } from './Button';

describe('Button', () => {
  it('renders with text', () => {
    render(<Button>点击</Button>);
    expect(screen.getByText('点击')).toBeInTheDocument();
  });
  
  it('calls onClick when clicked', async () => {
    const handleClick = jest.fn();
    const user = userEvent.setup();
    
    render(<Button onClick={handleClick}>点击</Button>);
    await user.click(screen.getByText('点击'));
    
    expect(handleClick).toHaveBeenCalledTimes(1);
  });
  
  it('shows loading state', () => {
    render(<Button loading>保存</Button>);
    
    expect(screen.getByRole('button')).toBeDisabled();
    expect(screen.getByRole('status')).toBeInTheDocument(); // Spinner
  });
  
  it('applies variant styles', () => {
    const { rerender } = render(<Button variant="primary">主按钮</Button>);
    expect(screen.getByRole('button')).toHaveClass('bg-primary-500');
    
    rerender(<Button variant="secondary">次按钮</Button>);
    expect(screen.getByRole('button')).toHaveClass('bg-gray-500');
  });
  
  it('is keyboard accessible', async () => {
    const handleClick = jest.fn();
    const user = userEvent.setup();
    
    render(<Button onClick={handleClick}>按钮</Button>);
    
    // Tab 聚焦
    await user.tab();
    expect(screen.getByRole('button')).toHaveFocus();
    
    // Enter 键触发
    await user.keyboard('{Enter}');
    expect(handleClick).toHaveBeenCalledTimes(1);
  });
});
```

### 4.2 Input 组件测试

```tsx
// src/components/Input/Input.test.tsx
import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { Input } from './Input';

describe('Input', () => {
  it('updates value on change', async () => {
    const handleChange = jest.fn();
    const user = userEvent.setup();
    
    render(<Input onChange={handleChange} placeholder="输入文本" />);
    
    const input = screen.getByPlaceholderText('输入文本');
    await user.type(input, 'Hello');
    
    expect(handleChange).toHaveBeenCalledTimes(5); // H, e, l, l, o
    expect(input).toHaveValue('Hello');
  });
  
  it('shows error message', () => {
    render(<Input error="必填字段" />);
    
    expect(screen.getByText('必填字段')).toBeInTheDocument();
    expect(screen.getByRole('textbox')).toHaveClass('border-error-500');
    expect(screen.getByRole('textbox')).toHaveAttribute('aria-invalid', 'true');
  });
  
  it('shows helper text', () => {
    render(<Input helperText="至少 8 个字符" />);
    expect(screen.getByText('至少 8 个字符')).toBeInTheDocument();
  });
  
  it('disables input', () => {
    render(<Input disabled />);
    expect(screen.getByRole('textbox')).toBeDisabled();
  });
});
```

### 4.3 Table 组件测试

```tsx
// src/components/Table/Table.test.tsx
import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { Table } from './Table';

const mockColumns = [
  { key: 'name', title: '名称', dataIndex: 'name' },
  { key: 'status', title: '状态', dataIndex: 'status' },
];

const mockData = [
  { id: 1, name: '数据源A', status: '已连接' },
  { id: 2, name: '数据源B', status: '未连接' },
];

describe('Table', () => {
  it('renders headers and rows', () => {
    render(<Table columns={mockColumns} dataSource={mockData} rowKey="id" />);
    
    // 验证表头
    expect(screen.getByText('名称')).toBeInTheDocument();
    expect(screen.getByText('状态')).toBeInTheDocument();
    
    // 验证数据
    expect(screen.getByText('数据源A')).toBeInTheDocument();
    expect(screen.getByText('已连接')).toBeInTheDocument();
  });
  
  it('sorts by column', async () => {
    const user = userEvent.setup();
    
    render(
      <Table 
        columns={mockColumns.map(col => ({ ...col, sortable: true }))} 
        dataSource={mockData} 
        rowKey="id" 
      />
    );
    
    // 点击表头排序
    await user.click(screen.getByText('名称'));
    
    // 验证排序后顺序
    const rows = screen.getAllByRole('row');
    expect(rows[1]).toHaveTextContent('数据源A');
    expect(rows[2]).toHaveTextContent('数据源B');
  });
  
  it('shows empty state', () => {
    render(<Table columns={mockColumns} dataSource={[]} rowKey="id" />);
    expect(screen.getByText('暂无数据')).toBeInTheDocument();
  });
});
```

## 五、API Mock（MSW）

### 5.1 配置 MSW

```tsx
// src/mocks/handlers.ts
import { http, HttpResponse } from 'msw';

export const handlers = [
  // 数据源列表
  http.get('/api/v1/data_sources', () => {
    return HttpResponse.json({
      items: [
        { id: 1, name: '淘宝店A', platform: 'taobao', status: 'connected' },
        { id: 2, name: '京东店B', platform: 'jd', status: 'disconnected' },
      ],
      total: 2,
      page: 1,
      pages: 1,
    });
  }),
  
  // 创建数据源
  http.post('/api/v1/data_sources', async ({ request }) => {
    const body = await request.json();
    return HttpResponse.json(
      { id: 3, ...body },
      { status: 201 }
    );
  }),
  
  // 删除数据源
  http.delete('/api/v1/data_sources/:id', () => {
    return new HttpResponse(null, { status: 204 });
  }),
];
```

## 七、可访问性测试

### 7.1 jest-axe 集成

```tsx
// src/components/Button/Button.a11y.test.tsx
import { render } from '@testing-library/react';
import { axe, toHaveNoViolations } from 'jest-axe';
import { Button } from './Button';

expect.extend(toHaveNoViolations);

describe('Button 可访问性', () => {
  it('无可访问性违规', async () => {
    const { container } = render(<Button>点击</Button>);
    const results = await axe(container);
    expect(results).toHaveNoViolations();
  });
  
  it('禁用状态无可访问性违规', async () => {
    const { container } = render(<Button disabled>禁用</Button>);
    const results = await axe(container);
    expect(results).toHaveNoViolations();
  });
  
  it('加载状态有 aria-live', () => {
    const { container } = render(<Button loading>加载中</Button>);
    expect(container.querySelector('[role="status"]')).toBeInTheDocument();
  });
});
```

### 7.2 Playwright 可访问性测试

```typescript
// tests/e2e/accessibility.spec.ts
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

test.describe('可访问性测试', () => {
  test('Dashboard 页面无可访问性违规', async ({ page }) => {
    await page.goto('/dashboard');
    
    const accessibilityScanResults = await new AxeBuilder({ page })
      .withTags(['wcag2a', 'wcag2aa'])
      .analyze();
    
    expect(accessibilityScanResults.violations).toEqual([]);
  });
  
  test('表单错误有正确的 ARIA 属性', async ({ page }) => {
    await page.goto('/data-sources/new');
    
    // 提交空表单触发验证错误
    await page.click('button[type="submit"]');
    
    // 验证 aria-invalid
    const nameInput = page.locator('[name="name"]');
    await expect(nameInput).toHaveAttribute('aria-invalid', 'true');
    
    // 验证 aria-describedby
    const errorId = await nameInput.getAttribute('aria-describedby');
    expect(errorId).toBeTruthy();
    await expect(page.locator(`#${errorId}`)).toContainText('必填字段');
  });
});
```

```yaml
# .github/workflows/frontend-test.yml

on: [push, pull_request]

jobs:
  unit-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      
      - name: 安装依赖
        run: cd frontend && npm ci
      
      - name: 运行单元测试
        run: npm test -- --coverage
      
      - name: 上传覆盖率
        uses: codecov/codecov-action@v3
  
  e2e-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
      
      - name: 安装依赖
        run: cd frontend && npm ci
      
      - name: 安装 Playwright
        run: npx playwright install --with-deps
      
      - name: 运行 E2E 测试
        run: npm run test:e2e
      
      - name: 上传测试报告
        if: always()
        uses: actions/upload-artifact@v3
        with:
          name: playwright-report
          path: playwright-report/
```
