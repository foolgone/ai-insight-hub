## 1. 数据库概述

### 1.1 技术选型

**PostgreSQL 14+ 的优势**：
- ✅ JSONB 支持（灵活存储非结构化数据）
- ✅ 全文搜索（中文支持）
- ✅ 向量扩展（pgvector）
- ✅ 分区表（大数据量优化）
- ✅ 行级安全（RLS）
- ✅ 成熟稳定

---

### 1.2 数据库规模估算

**假设场景**：中型企业，100 万条对话/月

| 表 | 预估行数（1年） | 单行大小 | 总大小 |
|-----|----------------|----------|--------|
| users | 1,000 | 0.5 KB | 0.5 MB |
| data_sources | 10,000 | 1 KB | 10 MB |
| raw_data | 12M | 2 KB | 24 GB |
| cleaned_data | 12M | 3 KB | 36 GB |
| analysis_results | 12M | 5 KB | 60 GB |
| knowledge_base | 50,000 | 4 KB | 200 MB |
| reports | 10,000 | 0.5 KB | 5 MB |
| **总计** | | | **~120 GB** |

**扩展**：支持 1000 万条对话（10 倍）→ 1.2 TB

## 3. 表结构设计

### 3.1 用户表 (users)

```sql
CREATE TABLE users (
    -- 主键
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- 认证信息
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    
    -- 用户信息
    name VARCHAR(100),
    avatar_url VARCHAR(500),
    
    -- 权限
    role VARCHAR(50) DEFAULT 'user' CHECK (role IN ('admin', 'user', 'viewer')),
    is_active BOOLEAN DEFAULT TRUE,
    is_verified BOOLEAN DEFAULT FALSE,
    
    -- 配额
    quota JSONB DEFAULT '{
        "max_datasources": 10,
        "max_monthly_records": 100000,
        "max_monthly_tokens": 1000000
    }',
    usage JSONB DEFAULT '{
        "datasources_count": 0,
        "monthly_records": 0,
        "monthly_tokens": 0
    }',
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    last_login_at TIMESTAMP,
    
    -- 软删除
    deleted_at TIMESTAMP
);

-- 索引
CREATE INDEX idx_users_email ON users(email) WHERE deleted_at IS NULL;
CREATE INDEX idx_users_role ON users(role) WHERE deleted_at IS NULL;
CREATE INDEX idx_users_created_at ON users(created_at);

-- 注释
COMMENT ON TABLE users IS '用户表';
COMMENT ON COLUMN users.quota IS '用户配额限制';
COMMENT ON COLUMN users.usage IS '当前使用情况';
```

### 3.2 数据源表 (data_sources)

```sql
CREATE TABLE data_sources (
    -- 主键
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- 基本信息
    name VARCHAR(255) NOT NULL,
    description TEXT,
    type VARCHAR(50) NOT NULL CHECK (type IN ('customer_service', 'sales', 'support', 'other')),
    
    -- 配置
    config JSONB DEFAULT '{}',
    -- config 示例:
    -- {
    --   "auto_clean": true,
    --   "auto_evaluate": true,
    --   "evaluation_dimensions": ["accuracy", "completeness"],
    --   "notification_email": "user@example.com"
    -- }
    
    -- 状态
    status VARCHAR(50) DEFAULT 'active' CHECK (status IN ('active', 'archived', 'processing', 'error')),
    
    -- 统计
    statistics JSONB DEFAULT '{
        "total_records": 0,
        "cleaned_records": 0,
        "quality_score_avg": 0,
        "last_processed_at": null
    }',
    
    -- 所属用户
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    deleted_at TIMESTAMP
);

-- 索引
CREATE INDEX idx_datasources_user_id ON data_sources(user_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_datasources_status ON data_sources(status) WHERE deleted_at IS NULL;
CREATE INDEX idx_datasources_type ON data_sources(type) WHERE deleted_at IS NULL;
CREATE INDEX idx_datasources_created_at ON data_sources(created_at);
CREATE INDEX idx_datasources_config ON data_sources USING GIN(config);

-- 注释
COMMENT ON TABLE data_sources IS '数据源表';
COMMENT ON COLUMN data_sources.config IS '数据源配置（JSON）';
COMMENT ON COLUMN data_sources.statistics IS '统计信息（JSON）';
```

### 3.3 原始数据表 (raw_data)

**分区表**（按月分区）

```sql
-- 主表
CREATE TABLE raw_data (
    -- 主键
    id UUID DEFAULT gen_random_uuid(),
    
    -- 所属数据源
    datasource_id UUID NOT NULL REFERENCES data_sources(id) ON DELETE CASCADE,
    
    -- 原始内容
    raw_content JSONB NOT NULL,
    -- raw_content 示例:
    -- {
    --   "text": "时间: 2025-01-15 10:30\n客户: 你好...",
    --   "format": "csv",
    --   "encoding": "utf-8"
    -- }
    
    -- 文件信息（如果是文件上传）
    file_info JSONB,
    -- file_info 示例:
    -- {
    --   "filename": "data.csv",
    --   "size": 1024000,
    --   "mime_type": "text/csv",
    --   "upload_url": "s3://..."
    -- }
    
    -- 处理状态
    status VARCHAR(50) DEFAULT 'pending' CHECK (status IN ('pending', 'processing', 'completed', 'failed')),
    error_message TEXT,
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT NOW(),
    processed_at TIMESTAMP,
    
    -- 分区键
    PRIMARY KEY (id, created_at)
) PARTITION BY RANGE (created_at);

-- 创建分区（示例：2025 年的月度分区）
CREATE TABLE raw_data_2025_01 PARTITION OF raw_data
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

CREATE TABLE raw_data_2025_02 PARTITION OF raw_data
    FOR VALUES FROM ('2025-02-01') TO ('2025-03-01');

-- ... 后续月份

-- 索引（会自动应用到所有分区）
CREATE INDEX idx_raw_data_datasource_id ON raw_data(datasource_id, created_at);
CREATE INDEX idx_raw_data_status ON raw_data(status, created_at);
CREATE INDEX idx_raw_data_created_at ON raw_data(created_at);

-- 注释
COMMENT ON TABLE raw_data IS '原始数据表（按月分区）';
```

### 3.4 清洗后数据表 (cleaned_data)

**分区表**（按月分区）

```sql
-- 主表
CREATE TABLE cleaned_data (
    -- 主键
    id UUID DEFAULT gen_random_uuid(),
    
    -- 关联
    datasource_id UUID NOT NULL REFERENCES data_sources(id) ON DELETE CASCADE,
    raw_data_id UUID NOT NULL,
    
    -- 结构化数据
    dialogue_id UUID NOT NULL,
    timestamp TIMESTAMP NOT NULL,
    user_message TEXT NOT NULL,
    ai_response TEXT NOT NULL,
    
    -- 元数据
    metadata JSONB DEFAULT '{}',
    -- metadata 示例:
    -- {
    --   "source": "zendesk",
    --   "session_id": "xxx",
    --   "user_id": "customer_123",
    --   "agent_id": "bot_456"
    -- }
    
    -- 质量评分
    quality_score FLOAT CHECK (quality_score >= 0 AND quality_score <= 100),
    issues TEXT[],  -- 数组：["missing_timestamp", "short_message"]
    
    -- 向量（用于相似度搜索）
    embedding VECTOR(1536),  -- 需要 pgvector 扩展
    
    -- Agent 信息
    agent_version VARCHAR(50),
    processing_duration FLOAT,  -- 秒
    token_usage JSONB,  -- {input: 100, output: 200}
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT NOW(),
    
    -- 分区键
    PRIMARY KEY (id, created_at)
) PARTITION BY RANGE (created_at);

-- 创建分区
CREATE TABLE cleaned_data_2025_01 PARTITION OF cleaned_data
    FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');

-- 索引
CREATE INDEX idx_cleaned_data_datasource_id ON cleaned_data(datasource_id, created_at);
CREATE INDEX idx_cleaned_data_dialogue_id ON cleaned_data(dialogue_id);
CREATE INDEX idx_cleaned_data_timestamp ON cleaned_data(timestamp);
CREATE INDEX idx_cleaned_data_quality_score ON cleaned_data(quality_score);

-- 向量索引（相似度搜索）
CREATE INDEX idx_cleaned_data_embedding ON cleaned_data 
    USING ivfflat (embedding vector_cosine_ops)
    WITH (lists = 100);

-- 全文搜索索引
CREATE INDEX idx_cleaned_data_fts ON cleaned_data 
    USING GIN (to_tsvector('chinese', user_message || ' ' || ai_response));

-- 注释
COMMENT ON TABLE cleaned_data IS '清洗后数据表（按月分区）';
COMMENT ON COLUMN cleaned_data.embedding IS '文本向量表示（1536 维）';
```

### 3.5 分析结果表 (analysis_results)

```sql
CREATE TABLE analysis_results (
    -- 主键
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- 关联
    datasource_id UUID NOT NULL REFERENCES data_sources(id) ON DELETE CASCADE,
    data_id UUID NOT NULL,  -- cleaned_data 的 id
    
    -- 分析类型
    analysis_type VARCHAR(50) NOT NULL CHECK (
        analysis_type IN ('quality', 'knowledge', 'insight', 'other')
    ),
    
    -- 结果
    result JSONB NOT NULL,
    -- result 示例（质量评估）:
    -- {
    --   "overall_score": 85,
    --   "dimensions": {
    --     "accuracy": {"score": 90, "issues": []},
    --     "completeness": {"score": 80, "issues": ["incomplete"]}
    --   },
    --   "needs_review": false
    -- }
    
    -- Agent 信息
    agent_name VARCHAR(100) NOT NULL,
    agent_version VARCHAR(50),
    ai_model VARCHAR(100),
    
    -- 性能指标
    confidence FLOAT CHECK (confidence >= 0 AND confidence <= 1),
    processing_duration FLOAT,
    token_usage JSONB,
    cost FLOAT,  -- 美元
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_analysis_datasource_id ON analysis_results(datasource_id, created_at);
CREATE INDEX idx_analysis_data_id ON analysis_results(data_id);
CREATE INDEX idx_analysis_type ON analysis_results(analysis_type);
CREATE INDEX idx_analysis_created_at ON analysis_results(created_at);
CREATE INDEX idx_analysis_result ON analysis_results USING GIN(result);

-- 注释
COMMENT ON TABLE analysis_results IS '分析结果表';
```

## 4. 索引设计

### 4.1 索引策略

**原则**：
1. 查询频繁的列建索引
2. WHERE/JOIN/ORDER BY 的列建索引
3. 避免过多索引（影响写性能）
4. 定期分析和优化

---

### 4.2 复合索引

```sql
-- 数据源 + 时间范围查询
CREATE INDEX idx_cleaned_data_ds_time 
    ON cleaned_data(datasource_id, created_at DESC);

-- 数据源 + 质量评分查询
CREATE INDEX idx_cleaned_data_ds_quality 
    ON cleaned_data(datasource_id, quality_score DESC)
    WHERE quality_score < 70;  -- 部分索引，只索引低质量数据

-- 用户 + 数据源状态查询
CREATE INDEX idx_datasources_user_status 
    ON data_sources(user_id, status)
    WHERE deleted_at IS NULL;
```

---

### 4.3 特殊索引

**GIN 索引（JSONB）**：
```sql
-- JSONB 字段索引
CREATE INDEX idx_datasources_config_gin 
    ON data_sources USING GIN(config);

-- 查询示例
SELECT * FROM data_sources 
WHERE config @> '{"auto_clean": true}';
```

**全文搜索索引**：
```sql
-- 安装中文分词扩展
CREATE EXTENSION zhparser;
CREATE TEXT SEARCH CONFIGURATION chinese (PARSER = zhparser);

-- 创建全文搜索索引
CREATE INDEX idx_cleaned_data_search 
    ON cleaned_data 
    USING GIN (
        to_tsvector('chinese', user_message || ' ' || ai_response)
    );

-- 查询示例
SELECT * FROM cleaned_data
WHERE to_tsvector('chinese', user_message || ' ' || ai_response)
```

## 5. 分区策略

### 5.1 为什么分区？

**优势**：
- ✅ 查询性能提升（只扫描相关分区）
- ✅ 维护方便（可以单独删除旧分区）
- ✅ 并行处理（多个分区并行扫描）

---

### 5.2 分区表管理

**自动创建分区的存储过程**：

```sql
CREATE OR REPLACE FUNCTION create_monthly_partitions(
    table_name TEXT,
    start_date DATE,
    end_date DATE
)
RETURNS VOID AS $$
DECLARE
    partition_date DATE;
    partition_name TEXT;
    start_range DATE;
    end_range DATE;
BEGIN
    partition_date := DATE_TRUNC('month', start_date);
    
    WHILE partition_date < end_date LOOP
        partition_name := table_name || '_' || TO_CHAR(partition_date, 'YYYY_MM');
        start_range := partition_date;
        end_range := partition_date + INTERVAL '1 month';
        
        -- 检查分区是否已存在
        IF NOT EXISTS (
            SELECT 1 FROM pg_tables WHERE tablename = partition_name
        ) THEN
            EXECUTE format(
                'CREATE TABLE %I PARTITION OF %I FOR VALUES FROM (%L) TO (%L)',
                partition_name, table_name, start_range, end_range
            );
            RAISE NOTICE '创建分区: %', partition_name;
        END IF;
        
        partition_date := partition_date + INTERVAL '1 month';
    END LOOP;
END;
$$ LANGUAGE plpgsql;

-- 使用：创建 2025 年的分区
SELECT create_monthly_partitions('raw_data', '2025-01-01', '2026-01-01');
SELECT create_monthly_partitions('cleaned_data', '2025-01-01', '2026-01-01');
SELECT create_monthly_partitions('audit_logs', '2025-01-01', '2026-01-01');
```

**自动删除旧分区**：

```sql
CREATE OR REPLACE FUNCTION drop_old_partitions(
    table_name TEXT,
    retention_months INT
)
RETURNS VOID AS $$
DECLARE
    partition_record RECORD;
    cutoff_date DATE;
BEGIN
    cutoff_date := DATE_TRUNC('month', NOW()) - (retention_months || ' months')::INTERVAL;
    
    FOR partition_record IN
        SELECT tablename FROM pg_tables
        WHERE tablename LIKE table_name || '_%'
    LOOP
        -- 提取分区日期
        DECLARE
            partition_date DATE;
        BEGIN
            partition_date := TO_DATE(
                SUBSTRING(partition_record.tablename FROM '\d{4}_\d{2}$'),
                'YYYY_MM'
            );
            
            IF partition_date < cutoff_date THEN
                EXECUTE format('DROP TABLE %I', partition_record.tablename);
                RAISE NOTICE '删除分区: %', partition_record.tablename;
            END IF;
        END;
    END LOOP;
END;
$$ LANGUAGE plpgsql;

-- 使用：删除 12 个月前的分区
SELECT drop_old_partitions('raw_data', 12);
```

## 7. 备份策略

### 7.1 备份方案

**1. 全量备份（每日）**：

```bash
#!/bin/bash
# backup_full.sh

DATE=$(date +%Y%m%d)
BACKUP_DIR=/backups
DB_NAME=aiinsight

pg_dump -U postgres $DB_NAME | gzip > $BACKUP_DIR/full_$DATE.sql.gz

# 保留 30 天
find $BACKUP_DIR -name "full_*.sql.gz" -mtime +30 -delete
```

**2. 增量备份（WAL 归档）**：

```sql
-- postgresql.conf
wal_level = replica
archive_mode = on
archive_command = 'cp %p /archive/%f'
```

```bash

```
