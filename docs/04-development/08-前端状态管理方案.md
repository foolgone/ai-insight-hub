## 1. 问题描述

### 当前问题

原有设计同时提到了 React Query 和 Zustand，但没有明确：

❌ 各自的使用场景  
❌ 什么数据应该放在哪里  
❌ 如何避免混用导致的状态混乱  

### 影响

- 团队协作时出现分歧
- 状态管理混乱
- 代码维护困难
- 可能出现状态不同步

## 2. 状态管理方案

### 2.1 核心原则

```
明确分工，各司其职：

📡 React Query → 服务端状态（远程数据）
🗃️ Zustand     → 客户端状态（本地状态）
```

---

### 2.2 状态分类

#### 📡 服务端状态（React Query）

**定义**：来自 API 的数据，真实源在服务器

**特点**：
- ✅ 需要异步获取
- ✅ 可能过期（需要刷新）
- ✅ 多个组件可能共享
- ✅ 需要缓存和同步

**示例**：
```javascript
// ✅ 服务端状态
- 数据源列表
- 数据源详情
- 清洗结果
- 评估报告
- 知识库 FAQ
- 用户配额信息
- 分析统计数据
```

---

#### 🗃️ 客户端状态（Zustand）

**定义**：只存在于浏览器的状态，服务器不关心

**特点**：
- ✅ 同步数据（立即可用）
- ✅ 不需要持久化到服务器
- ✅ 纯 UI 交互状态
- ✅ 跨组件共享（非服务端数据）

**示例**：
```javascript
// ✅ 客户端状态
- 当前登录用户信息（Token）
- UI 主题（亮色/暗色）
- 侧边栏展开/收起
- 当前选中的 Tab
- 表单草稿（未提交）
- 通知提示（Toast）
- Modal 打开/关闭状态
```

---

### 2.3 决策树

```
需要管理的状态
    │
    ├─ 数据来自 API？
    │   ├─ 是 → React Query 📡
    │   └─ 否 → 继续判断
    │
    ├─ 需要跨多个组件共享？
    │   ├─ 是 → Zustand 🗃️
    │   └─ 否 → useState（组件内部）
    │
    └─ 纯 UI 交互状态？
        ├─ 是 → Zustand 🗃️
        └─ 否 → Context（特殊场景）
```

#### 全局配置

```javascript
// src/lib/queryClient.js
import { QueryClient } from '@tanstack/react-query';

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 5 * 60 * 1000,      // 5 分钟内数据视为新鲜
      gcTime: 10 * 60 * 1000,        // v5：cacheTime 已改名 gcTime，10 分钟后回收缓存
      refetchOnWindowFocus: false,   // 窗口聚焦时不自动刷新
      // 只重试网络/5xx 类错误；4xx 直接失败，避免无意义重试
      retry: (failureCount, error) => {
        const status = error?.response?.status;
        if (status && status >= 400 && status < 500) return false;
        return failureCount < 3;
      },
      retryDelay: (attemptIndex) =>   // 指数退避
        Math.min(1000 * 2 ** attemptIndex, 30000),
    },
    mutations: {
      retry: 1,                       // 写操作只重试 1 次
    },
  },
});
```

#### 使用示例

**查询数据**：

```javascript
// src/features/datasource/api.js
import api from '@/lib/api';

// 注意：axios 返回的是 AxiosResponse，这里统一解包出 data，
// 让 React Query 的 data 就是业务数据（拦截器已把键转成 camelCase）
export const datasourceApi = {
  // 获取列表
  fetchList: async (params) => {
    const res = await api.get('/datasources', { params });
    return res.data;
  },

  // 获取详情
  fetchById: async (id) => {
    const res = await api.get(`/datasources/${id}`);
    return res.data;
  },

  // 创建
  create: async (data) => {
    const res = await api.post('/datasources', data);
    return res.data;
  },

  // 更新
  update: async (id, data) => {
    const res = await api.patch(`/datasources/${id}`, data);
    return res.data;
  },

  // 删除
  delete: async (id) => {
    const res = await api.delete(`/datasources/${id}`);
    return res.data;
  },
};
```

```javascript
// src/features/datasource/hooks.js
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { datasourceApi } from './api';

// 查询列表
export function useDataSources(params = {}) {
  return useQuery({
    queryKey: ['datasources', params],
    queryFn: () => datasourceApi.fetchList(params),
  });
}

// 查询详情（options 用于 select / staleTime 等，签名与列表 hook 保持一致）
export function useDataSource(id, options = {}) {
  return useQuery({
    ...options,
    queryKey: ['datasources', id],
    queryFn: () => datasourceApi.fetchById(id),
    enabled: Boolean(id) && (options.enabled ?? true),
  });
}

// 创建
export function useCreateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: datasourceApi.create,
    onSuccess: () => {
      // 创建成功后，刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 更新
export function useUpdateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: ({ id, data }) => datasourceApi.update(id, data),
    onSuccess: (data, variables) => {
      // 更新成功后，刷新该数据源
      queryClient.invalidateQueries({ queryKey: ['datasources', variables.id] });
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 删除
export function useDeleteDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: datasourceApi.delete,
    onSuccess: () => {
      // 删除成功后，刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}
```

```jsx
// src/pages/DataSources/DataSourceList.jsx
import { useState } from 'react';
import toast from 'react-hot-toast';

import {
  useDataSources,
  useCreateDataSource,
  useDeleteDataSource,
} from '@/features/datasource/hooks';

function DataSourceList() {
  const [showCreateModal, setShowCreateModal] = useState(false);

  // 查询数据
  const { data, isLoading, error } = useDataSources({
    page: 1,
    limit: 20,
  });

  // 创建
  const createMutation = useCreateDataSource();

  // 删除
  const deleteMutation = useDeleteDataSource();

  const handleCreate = async (formData) => {
    try {
      await createMutation.mutateAsync(formData);
      setShowCreateModal(false);
      toast.success('创建成功');
    } catch (err) {
      toast.error(err.message);
    }
  };

  const handleDelete = async (id) => {
    if (!window.confirm('确定删除？')) return;

    try {
      await deleteMutation.mutateAsync(id);
      toast.success('删除成功');
    } catch (err) {
      toast.error(err.message);
    }
  };

  if (isLoading) return <LoadingSpinner />;
  if (error) return <ErrorMessage message={error.message} />;

  return (
    <div>
      <button type="button" onClick={() => setShowCreateModal(true)}>
        创建数据源
      </button>

      {showCreateModal && (
        <CreateDataSourceModal
          onClose={() => setShowCreateModal(false)}
          onSubmit={handleCreate}
        />
      )}

      {(data?.items ?? []).map((item) => (
        <DataSourceCard
          key={item.id}
          dataSource={item}
          onDelete={() => handleDelete(item.id)}
        />
      ))}
    </div>
  );
}
```

#### Store 定义

Token 存储口径（**T4，2026-10-26 已落地**）：

- **access_token**：只存内存（`src/lib/token.js`），不落 Web Storage；
- **refresh_token**：由后端写入 **HttpOnly Cookie**（默认名 `aiih_refresh`，`Path=/api/v1/auth`，`SameSite=lax`），JS 不可读，前端不持久化；
- 「记住我」只决定**用户信息**写 `localStorage` 还是 `sessionStorage`（key `aiih.auth`，内容仅 `{user, remember}`）。

```javascript
// src/lib/token.js（节选）
let accessToken = null
export function setAccessToken(token) {
  accessToken = token || null
}
export function getAccessToken() {
  return accessToken
}
export function clearAccessToken() {
  accessToken = null
}
```

```javascript
// src/lib/authStorage.js（节选）
export const AUTH_STORAGE_KEY = 'aiih.auth'

export function saveAuth({ user, remember = false }) {
  clearAuth()
  const target = remember ? window.localStorage : window.sessionStorage
  target.setItem(AUTH_STORAGE_KEY, JSON.stringify({ user: user || null, remember }))
}

export function loadAuth() {
  // 依次读 localStorage / sessionStorage，返回 { user, remember } 或 null
}
```

刷新流程（`src/api/client.js`）：业务请求 401 → 若本地仍有用户会话 → `POST /auth/refresh`（`withCredentials: true`，浏览器自动带 Cookie）→ **单飞**刷新 → 重放原请求；刷新失败则清会话并跳登录页。

> ✅ **安全收益（相对旧方案）**：XSS 读不到 refresh token，长效凭证只在 HttpOnly Cookie 中；刷新接口靠 `SameSite=Lax` 阻断跨站携带 Cookie。
> ⚠️ **剩余风险**：access_token 仍存内存（XSS 可在最多 60 分钟有效期内窃取），需靠 CSP、输入转义、禁用 `dangerouslySetInnerHTML` 收敛；若未来跨站部署，需评估 `SameSite=None; Secure` + CSRF Token。

```javascript
// src/stores/uiStore.js
import { create } from 'zustand';

export const useUIStore = create((set) => ({
  // 侧边栏
  sidebarOpen: true,
  toggleSidebar: () => set((state) => ({ sidebarOpen: !state.sidebarOpen })),

  // 通知
  notifications: [],

      notifications: [...state.notifications, {
        id: Date.now(),
        ...notification,
      }],
    })),
  removeNotification: (id) =>
    set((state) => ({
      notifications: state.notifications.filter((n) => n.id !== id),
    })),
  
  // Modal
  activeModal: null,
  openModal: (modalName) => set({ activeModal: modalName }),
  closeModal: () => set({ activeModal: null }),
}));
```

```javascript
// src/stores/formStore.js
import { create } from 'zustand';

// 表单草稿（防止用户误关闭丢失数据）
export const useFormStore = create((set, get) => ({
  drafts: {},
  
  saveDraft: (formName, data) => {
    set((state) => ({
      drafts: {
        ...state.drafts,
        [formName]: {
          data,
          savedAt: new Date().toISOString(),
        },
      },
    }));
  },
  
  getDraft: (formName) => {
    return get().drafts[formName]?.data;
  },
  
  clearDraft: (formName) => {
    set((state) => {
      const { [formName]: _, ...rest } = state.drafts;
      return { drafts: rest };
    });
  },
}));
```

```jsx
      description: '',
```

### 4.2 避免混用

#### ❌ 错误做法

```javascript

const useStore = create((set) => ({
```

```javascript
// ✅ React Query：服务端数据
```

### 4.3 性能优化

**React Query**：

```jsx
// 1. 预取数据：鼠标悬停时预取详情
function DataSourceCardWithPrefetch({ id, dataSource, onDelete }) {
  const queryClient = useQueryClient();

  const handleMouseEnter = () => {
    queryClient.prefetchQuery({
      queryKey: ['datasources', id],
      queryFn: () => datasourceApi.fetchById(id),
    });
  };

  return (
    <DataSourceCard
      dataSource={dataSource}
      onDelete={onDelete}
      onMouseEnter={handleMouseEnter}
    />
  );
}

// 2. 乐观更新（带失败回滚）
function useUpdateDataSourceOptimistic(id) {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (newData) => datasourceApi.update(id, newData),
    onMutate: async (newData) => {
      // 取消正在进行的查询，避免旧响应覆盖乐观更新
      await queryClient.cancelQueries({ queryKey: ['datasources', id] });

      // 快照旧数据
      const previous = queryClient.getQueryData(['datasources', id]);

      // 乐观更新
      queryClient.setQueryData(['datasources', id], (old) => ({
        ...old,
        ...newData,
      }));

      return { previous };
    },
    onError: (_err, _newData, context) => {
      // 出错时回滚
      if (context?.previous) {
        queryClient.setQueryData(['datasources', id], context.previous);
      }
    },
    onSettled: () => {
      // 无论成败都重新拉一次，保证与服务端一致
      queryClient.invalidateQueries({ queryKey: ['datasources', id] });
    },
  });
}

// 3. 选择性订阅（只订阅需要的字段，减少重渲染）
const { data } = useDataSource(id, {
  select: (detail) => ({
    name: detail.name,
    type: detail.type,
  }),
});
```

**Zustand**：

```javascript
// 1. 选择器（避免不必要的重渲染）
// ❌ 整个 store 变化都会重渲染
const store = useUIStore();

// ✅ 只有 sidebarOpen 变化才重渲染

// 2. 一次取多个字段：用 shallow 做浅比较
import { shallow } from 'zustand/shallow';

  (state) => ({ user: state.user, isAuthenticated: state.isAuthenticated }),

// ✅ 按功能拆分
useAuthStore()
useUIStore()
useFormStore()

// ❌ 单个大 Store

```
