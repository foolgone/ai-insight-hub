## 1. 问题描述

### 当前问题

原有设计同时提到了 React Query 和 Zustand，但没有明确：

❌ 各自的使用场景  
❌ 什么数据应该放在哪里  
❌ 如何避免混用导致的状态混乱  

### 影响

- 团队协作时出现分歧
- 状态管理混乱
- 代码维护困难
- 可能出现状态不同步

## 2. 状态管理方案

### 2.1 核心原则

```
明确分工，各司其职：

📡 React Query → 服务端状态（远程数据）
🗃️ Zustand     → 客户端状态（本地状态）
```

---

### 2.2 状态分类

#### 📡 服务端状态（React Query）

**定义**：来自 API 的数据，真实源在服务器

**特点**：
- ✅ 需要异步获取
- ✅ 可能过期（需要刷新）
- ✅ 多个组件可能共享
- ✅ 需要缓存和同步

**示例**：
```javascript
// ✅ 服务端状态
- 数据源列表
- 数据源详情
- 清洗结果
- 评估报告
- 知识库 FAQ
- 用户配额信息
- 分析统计数据
```

---

#### 🗃️ 客户端状态（Zustand）

**定义**：只存在于浏览器的状态，服务器不关心

**特点**：
- ✅ 同步数据（立即可用）
- ✅ 不需要持久化到服务器
- ✅ 纯 UI 交互状态
- ✅ 跨组件共享（非服务端数据）

**示例**：
```javascript
// ✅ 客户端状态
- 当前登录用户信息（Token）
- UI 主题（亮色/暗色）
- 侧边栏展开/收起
- 当前选中的 Tab
- 表单草稿（未提交）
- 通知提示（Toast）
- Modal 打开/关闭状态
```

---

### 2.3 决策树

```
需要管理的状态
    │
    ├─ 数据来自 API？
    │   ├─ 是 → React Query 📡
    │   └─ 否 → 继续判断
    │
    ├─ 需要跨多个组件共享？
    │   ├─ 是 → Zustand 🗃️
    │   └─ 否 → useState（组件内部）
    │
    └─ 纯 UI 交互状态？
        ├─ 是 → Zustand 🗃️
        └─ 否 → Context（特殊场景）
```

## 3. 实现指南

### 3.1 React Query 配置

#### 安装

```bash
npm install @tanstack/react-query @tanstack/react-query-devtools
```

#### 全局配置

```javascript
// src/lib/queryClient.js
import { QueryClient } from '@tanstack/react-query';

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 5 * 60 * 1000,      // 5 分钟内数据视为新鲜
      cacheTime: 10 * 60 * 1000,     // 10 分钟后清除缓存
      refetchOnWindowFocus: false,    // 窗口聚焦时不自动刷新
      retry: 3,                       // 失败时重试 3 次
      retryDelay: (attemptIndex) =>   // 指数退避
        Math.min(1000 * 2 ** attemptIndex, 30000),
    },
    mutations: {
      retry: 1,                       // 写操作只重试 1 次
    },
  },
});
```

#### 使用示例

**查询数据**：

```javascript
// src/features/datasource/api.js
import api from '@/lib/api';

export const datasourceApi = {
  // 获取列表
  fetchList: async (params) => {
    return api.get('/datasources', { params });
  },
  
  // 获取详情
  fetchById: async (id) => {
    return api.get(`/datasources/${id}`);
  },
  
  // 创建
  create: async (data) => {
    return api.post('/datasources', data);
  },
  
  // 更新
  update: async (id, data) => {
    return api.patch(`/datasources/${id}`, data);
  },
  
  // 删除
  delete: async (id) => {
    return api.delete(`/datasources/${id}`);
  },
};
```

```javascript
// src/features/datasource/hooks.js
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { datasourceApi } from './api';

// 查询列表
export function useDataSources(params = {}) {
  return useQuery({
    queryKey: ['datasources', params],
    queryFn: () => datasourceApi.fetchList(params),
  });
}

// 查询详情
export function useDataSource(id) {
  return useQuery({
    queryKey: ['datasources', id],
    queryFn: () => datasourceApi.fetchById(id),
    enabled: !!id,  // 只有 id 存在时才查询
  });
}

// 创建
export function useCreateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: datasourceApi.create,
    onSuccess: () => {
      // 创建成功后，刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 更新
export function useUpdateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: ({ id, data }) => datasourceApi.update(id, data),
    onSuccess: (data, variables) => {
      // 更新成功后，刷新该数据源
      queryClient.invalidateQueries({ queryKey: ['datasources', variables.id] });
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 删除
export function useDeleteDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: datasourceApi.delete,
    onSuccess: () => {
      // 删除成功后，刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}
```

**在组件中使用**：

```jsx
// src/pages/DataSources/DataSourceList.jsx
import { useDataSources, useCreateDataSource, useDeleteDataSource } from '@/features/datasource/hooks';

function DataSourceList() {
  // 查询数据
  const { data, isLoading, error, refetch } = useDataSources({
    page: 1,
    limit: 20,
  });
  
  // 创建
  const createMutation = useCreateDataSource();
  
  // 删除
  const deleteMutation = useDeleteDataSource();
  
  const handleCreate = async (formData) => {
    try {
      await createMutation.mutateAsync(formData);
      // 成功提示
      toast.success('创建成功');
    } catch (error) {
      // 错误提示
      toast.error(error.message);
    }
  };
  
  const handleDelete = async (id) => {
    if (!confirm('确定删除？')) return;
    
    try {
      await deleteMutation.mutateAsync(id);
      toast.success('删除成功');
    } catch (error) {
      toast.error(error.message);
    }
  };
  
  if (isLoading) return <LoadingSpinner />;
  if (error) return <ErrorMessage message={error.message} />;
  
  return (
    <div>
      <button onClick={() => setShowCreateModal(true)}>
        创建数据源
      </button>
      
      {data?.items.map(item => (
        <DataSourceCard
          key={item.id}
          dataSource={item}
          onDelete={() => handleDelete(item.id)}
        />
      ))}
    </div>
  );
}
```

```javascript
// src/stores/authStore.js
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

export const useAuthStore = create(
  persist(
    (set, get) => ({
      // 状态
      user: null,
      token: null,
      isAuthenticated: false,
      
      // Actions
      login: (user, token) => {
        set({
          user,
          token,
          isAuthenticated: true,
        });
      },
      
      logout: () => {
        set({
          user: null,
          token: null,
          isAuthenticated: false,
        });
        // 清除本地存储
        localStorage.removeItem('access_token');
      },
      
      updateUser: (userData) => {
        set((state) => ({
          user: { ...state.user, ...userData },
        }));
      },
    }),
    {
      name: 'auth-storage',  // localStorage key
      // 只持久化部分字段
      partialize: (state) => ({
        user: state.user,
        token: state.token,
        isAuthenticated: state.isAuthenticated,
      }),
    }
  )
);
```

```javascript
// src/stores/uiStore.js
import { create } from 'zustand';

export const useUIStore = create((set) => ({
  // 侧边栏
  sidebarOpen: true,
  toggleSidebar: () => set((state) => ({ sidebarOpen: !state.sidebarOpen })),
  
  // 主题
  theme: 'light',
  setTheme: (theme) => set({ theme }),
  
  // 通知
  notifications: [],
  addNotification: (notification) =>
    set((state) => ({
      notifications: [...state.notifications, {
        id: Date.now(),
        ...notification,
      }],
    })),
  removeNotification: (id) =>
    set((state) => ({
      notifications: state.notifications.filter((n) => n.id !== id),
    })),
  
  // Modal
  activeModal: null,
  openModal: (modalName) => set({ activeModal: modalName }),
  closeModal: () => set({ activeModal: null }),
}));
```

```javascript
// src/stores/formStore.js
import { create } from 'zustand';

// 表单草稿（防止用户误关闭丢失数据）
export const useFormStore = create((set, get) => ({
  drafts: {},
  
  saveDraft: (formName, data) => {
    set((state) => ({
      drafts: {
        ...state.drafts,
        [formName]: {
          data,
          savedAt: new Date().toISOString(),
        },
      },
    }));
  },
  
  getDraft: (formName) => {
    return get().drafts[formName]?.data;
  },
  
  clearDraft: (formName) => {
    set((state) => {
      const { [formName]: _, ...rest } = state.drafts;
      return { drafts: rest };
    });
  },
}));
```

```jsx
import { useFormStore } from '@/stores/formStore';
import { useCreateDataSource } from '@/features/datasource/hooks';

function CreateDataSourceForm() {
  const { saveDraft, getDraft, clearDraft } = useFormStore();
  const createMutation = useCreateDataSource();
  
  // 加载草稿
  const [formData, setFormData] = useState(() => {
    return getDraft('createDataSource') || {
      name: '',
      type: 'customer_service',
      description: '',
    };
  });
  
  // 自动保存草稿
  useEffect(() => {
    const timer = setTimeout(() => {
      saveDraft('createDataSource', formData);
    }, 1000);
    
    return () => clearTimeout(timer);
  }, [formData, saveDraft]);
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    try {
      await createMutation.mutateAsync(formData);
      // 提交成功后清除草稿
      clearDraft('createDataSource');
      toast.success('创建成功');
    } catch (error) {
      toast.error(error.message);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      {/* 表单字段 */}
    </form>
  );
}
```

### 4.1 命名规范

**React Query**：

```javascript
// ✅ 使用 use 前缀 + 资源名
useDataSources()      // 列表
useDataSource(id)     // 详情
useCreateDataSource() // 创建
useUpdateDataSource() // 更新
useDeleteDataSource() // 删除

// Query Key 使用数组
['datasources']                    // 所有数据源
['datasources', { page: 1 }]      // 带参数
['datasources', id]                // 单个数据源
['datasources', id, 'analysis']    // 嵌套资源
```

**Zustand**：

```javascript
// ✅ Store 命名：use + 名称 + Store
useAuthStore()
useUIStore()
useFormStore()

// ✅ Action 命名：动词 + 名词
login()
logout()
setTheme()
toggleSidebar()
addNotification()
```

### 4.2 避免混用

#### ❌ 错误做法

```javascript
// ❌ 用 Zustand 存储服务端数据
const useStore = create((set) => ({
  datasources: [],
  fetchDataSources: async () => {
    const data = await api.get('/datasources');
    set({ datasources: data });
  },
}));
// 问题：没有缓存、没有自动刷新、没有错误处理

// ❌ 用 React Query 存储 UI 状态
const { data: sidebarOpen } = useQuery({
  queryKey: ['sidebarOpen'],
  queryFn: () => true,
});
// 问题：UI 状态不应该用异步查询
```

#### ✅ 正确做法

```javascript
// ✅ React Query：服务端数据
const { data: datasources } = useDataSources();

// ✅ Zustand：UI 状态
const { sidebarOpen, toggleSidebar } = useUIStore();
```

### 4.3 性能优化

**React Query**：

```javascript
// 1. 预取数据
const queryClient = useQueryClient();

// 鼠标悬停时预取详情
<DataSourceCard
  onMouseEnter={() => {
    queryClient.prefetchQuery({
      queryKey: ['datasources', id],
      queryFn: () => datasourceApi.fetchById(id),
    });
  }}
/>

// 2. 乐观更新
const updateMutation = useMutation({
  mutationFn: datasourceApi.update,
  onMutate: async (newData) => {
    // 取消正在进行的查询
    await queryClient.cancelQueries({ queryKey: ['datasources', id] });
    
    // 快照旧数据
    const previous = queryClient.getQueryData(['datasources', id]);
    
    // 乐观更新
    queryClient.setQueryData(['datasources', id], newData);
    
    return { previous };
  },
  onError: (err, newData, context) => {
    // 出错时回滚
    queryClient.setQueryData(['datasources', id], context.previous);
  },
});

// 3. 选择性订阅
const { data } = useDataSource(id, {
  select: (data) => ({
    // 只选择需要的字段
    name: data.name,
    type: data.type,
  }),
});
```

**Zustand**：

```javascript
// 1. 选择器（避免不必要的重渲染）
// ❌ 整个 store 变化都会重渲染
const store = useUIStore();

// ✅ 只有 sidebarOpen 变化才重渲染
const sidebarOpen = useUIStore((state) => state.sidebarOpen);

// 2. 使用 shallow 比较（对象/数组）
import { shallow } from 'zustand/shallow';

const { user, theme } = useAuthStore(
  (state) => ({ user: state.user, theme: state.theme }),
  shallow
);

// 3. 拆分 Store（避免单个大 Store）
// ✅ 按功能拆分
useAuthStore()
useUIStore()
useFormStore()

// ❌ 单个大 Store
useAppStore()  // 包含所有状态
```
