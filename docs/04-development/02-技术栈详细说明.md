// 修改
function useCreateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: createDataSource,
    onSuccess: () => {
      // 刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 使用
function DataSourceList() {
  const { data, isLoading, error } = useDataSources();
  const createMutation = useCreateDataSource();

    createMutation.mutate(data);
  };
  
  if (isLoading) return <Spinner />;
  if (error) return <Error message={error.message} />;
  
  return <List data={data} onCreate={handleCreate} />;

### 4.2 Redis 7+

**用途**：
1. **缓存**（热点数据）
2. **消息队列**（Celery Broker）
3. **会话存储**（Session）
4. **限流**（Rate Limiting）

```python
import redis.asyncio as redis
from typing import Optional
import json

class RedisCache:
    def __init__(self, url: str):
        self.client = redis.from_url(url)
    
    async def get(self, key: str) -> Optional[dict]:
        """获取缓存"""
        data = await self.client.get(key)
        return json.loads(data) if data else None
    
    async def set(
        self,
        key: str,
        value: dict,
        ttl: int = 3600
    ):
        """设置缓存"""
        await self.client.setex(
            key,
            ttl,
            json.dumps(value)
        )
    
    async def delete(self, key: str):
        """删除缓存"""
        await self.client.delete(key)

# 使用
cache = RedisCache("redis://localhost:6379/0")

# 缓存数据源
await cache.set(f"datasource:{id}", datasource_data, ttl=600)

# 读取缓存
cached = await cache.get(f"datasource:{id}")
```

### 4.3 PostgreSQL + pgvector（向量搜索）

**用途**：向量数据库（知识库相似度检索）

**优势**：
- ✅ 一体化部署（无需额外服务）
- ✅ 支持 ACID 事务
- ✅ 可直接 JOIN 关联查询
- ✅ 维护简单

```python
from sqlalchemy import text
import numpy as np

# 安装扩展（仅需一次）
await db.execute("CREATE EXTENSION IF NOT EXISTS vector")

# 插入带向量的数据
from sentence_transformers import SentenceTransformer
model = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')

question = "iPhone 14 多少钱？"
embedding = model.encode(question).tolist()

await db.execute(
    """
    INSERT INTO knowledge_base (question, answer, embedding)
    VALUES (:question, :answer, :embedding)
    """,
    {
        "question": question,
        "answer": "iPhone 14 目前的价格是 5999 元起",
        "embedding": embedding
    }
)

# 相似度搜索（余弦距离）
query_embedding = model.encode("iPhone 14 价格").tolist()

results = await db.fetch_all(
    """
    SELECT question, answer,
           embedding <=> :query_embedding AS distance
    FROM knowledge_base
    ORDER BY distance
    LIMIT 5
    """,
    {"query_embedding": query_embedding}
)

# 返回最相似的 5 个结果
for row in results:
    print(f"问题: {row['question']}")
    print(f"答案: {row['answer']}")
    print(f"相似度: {1 - row['distance']:.2%}\n")
```

**索引优化**：
```sql
-- IVFFlat 索引（适合大规模数据）
CREATE INDEX idx_kb_embedding ON knowledge_base 
    USING ivfflat (embedding vector_cosine_ops)
    WITH (lists = 100);

-- HNSW 索引（更快，但占用更多内存）
CREATE INDEX idx_kb_embedding_hnsw ON knowledge_base 
    USING hnsw (embedding vector_cosine_ops);
```

### 5.2 LangGraph 0.0.20+

from langgraph.graph import StateGraph, END
from typing import TypedDict

# 定义状态
class ProcessState(TypedDict):

    cleaned_data: dict

    should_review,

        "extract": "extract"

)

## 9. 技术选型理由

### 9.1 为什么选 FastAPI？

✅ **优势**：
- 性能接近 Go/Node.js
- 自动生成文档
- 类型安全
- 异步原生支持
- 学习曲线平缓

❌ **劣势**：
- 生态不如 Django 成熟
- 没有内置 Admin

**对比**：
| 框架 | 性能 | 文档生成 | 异步 | 学习难度 |
|------|------|----------|------|----------|
| FastAPI | ⭐⭐⭐⭐⭐ | ✅ | ✅ | 低 |
| Django | ⭐⭐⭐ | ❌ | 部分 | 中 |
| Flask | ⭐⭐⭐⭐ | ❌ | ❌ | 低 |

---

### 9.2 为什么选 LangChain？

✅ **优势**：
- AI 应用开发的事实标准
- 丰富的集成（50+ LLMs）
- 活跃的社区
- LangSmith 可观测性
- LangGraph 工作流编排

❌ **劣势**：
- 版本迭代快
- API 有时不稳定

**对比**：
| 框架 | 成熟度 | 生态 | 可观测性 | 学习曲线 |
|------|--------|------|----------|----------|
| LangChain | ⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 中 |
| LlamaIndex | ⭐⭐⭐⭐ | ⭐⭐⭐ | ⭐⭐⭐ | 中 |
| Semantic Kernel | ⭐⭐⭐ | ⭐⭐ | ⭐⭐ | 低 |

---

### 9.3 为什么选 Claude？

✅ **优势**：
- 长上下文（200K tokens）
- 更准确（幻觉少）
- 更安全（内置安全机制）
- API 稳定

**对比**（准确性）：
```
Claude 3.5 Sonnet > GPT-4 Turbo > Claude 3 Opus > GPT-3.5
```

**成本对比**（每百万 tokens）：
| 模型 | 输入 | 输出 |
|------|------|------|
| Claude 3.5 Sonnet | $3 | $15 |
| GPT-4 Turbo | $10 | $30 |
| Claude Haiku | $0.8 | $4 |
| GPT-3.5 Turbo | $0.5 | $1.5 |
