```javascript
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

function useDataSources() {
  return useQuery({

    refetchOnWindowFocus: true,
  });
}

// 修改
function useCreateDataSource() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: createDataSource,
    onSuccess: () => {
      // 刷新列表
      queryClient.invalidateQueries({ queryKey: ['datasources'] });
    },
  });
}

// 使用
function DataSourceList() {
  const { data, isLoading, error } = useDataSources();
  const createMutation = useCreateDataSource();
  
  const handleCreate = (data) => {
    createMutation.mutate(data);
  };
  
  if (isLoading) return <Spinner />;
  if (error) return <Error message={error.message} />;
  
  return <List data={data} onCreate={handleCreate} />;
}
```

### 4.2 Redis 7+

**用途**：

2. **消息队列**（Celery Broker）
3. **会话存储**（Session）

```python
import redis.asyncio as redis
from typing import Optional
import json

class RedisCache:
    def __init__(self, url: str):
        self.client = redis.from_url(url)
    
    async def get(self, key: str) -> Optional[dict]:
        """获取缓存"""
        data = await self.client.get(key)
        return json.loads(data) if data else None
    
    async def set(
        self,
        key: str,
        value: dict,
        ttl: int = 3600
    ):
        """设置缓存"""
        await self.client.setex(
            key,
            ttl,
            json.dumps(value)
        )
    
    async def delete(self, key: str):
        """删除缓存"""
        await self.client.delete(key)

# 使用
cache = RedisCache("redis://localhost:6379/0")

# 缓存数据源
await cache.set(f"datasource:{id}", datasource_data, ttl=600)

# 读取缓存
cached = await cache.get(f"datasource:{id}")
```

### 4.3 PostgreSQL + pgvector（向量搜索）

**用途**：向量数据库（知识库相似度检索）

**优势**：
- ✅ 一体化部署（无需额外服务）
- ✅ 支持 ACID 事务
- ✅ 可直接 JOIN 关联查询
- ✅ 维护简单

```python
from sqlalchemy import text
import numpy as np

# 安装扩展（仅需一次）
await db.execute("CREATE EXTENSION IF NOT EXISTS vector")

# 插入带向量的数据
from sentence_transformers import SentenceTransformer

# 注意：该模型输出 384 维向量，列类型必须是 VECTOR(384)（不是 1536）
model = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')

question = "iPhone 14 多少钱？"
embedding = model.encode(question).tolist()

await db.execute(
    """
    INSERT INTO knowledge_base (question, answer, embedding)
    VALUES (:question, :answer, :embedding)
    """,
    {
        "question": question,
        "answer": "iPhone 14 目前的价格是 5999 元起",
        "embedding": embedding
    }
)

# 相似度搜索（余弦距离）
query_embedding = model.encode("iPhone 14 价格").tolist()

results = await db.fetch_all(
    """
    SELECT question, answer,
           embedding <=> :query_embedding AS distance
    FROM knowledge_base
    ORDER BY distance
    LIMIT 5
    """,
    {"query_embedding": query_embedding}
)

# 返回最相似的 5 个结果
for row in results:
    print(f"问题: {row['question']}")
    print(f"答案: {row['answer']}")
    print(f"相似度: {1 - row['distance']:.2%}\n")
```

### 5.2 LangGraph 0.0.20+

**用途**：复杂工作流编排

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict

# 定义状态
class ProcessState(TypedDict):
    datasource_id: str
    raw_data: dict
    cleaned_data: dict
    analysis_result: dict
    status: str

# 创建图
workflow = StateGraph(ProcessState)

# 添加节点
workflow.add_node("clean", cleaning_agent)
workflow.add_node("analyze", analysis_agent)
workflow.add_node("extract", extraction_agent)

# 添加边
workflow.add_edge("clean", "analyze")
workflow.add_edge("analyze", "extract")

# 条件分支
def should_review(state):
    if state["analysis_result"]["confidence"] < 0.7:
        return "review"
    return "extract"

workflow.add_conditional_edges(
    "analyze",
    should_review,
    {
        "review": "human_review",
        "extract": "extract"
    }
)
```

### 5.4 可观测性（自建，不使用 LangSmith）

**方案**：Loguru 结构化日志 + 数据库 LLM 调用记录 + Prometheus 指标，三者组合替代 SaaS 追踪。

```python
from loguru import logger
from app.core.metrics import LLM_TOKENS_TOTAL, LLM_COST_USD_TOTAL, LLM_LATENCY

logger.bind(
    agent="cleaning_agent",
    task_id=task_id,
    datasource_id=datasource_id,
    model=settings.DEFAULT_LLM_MODEL,
    run_name="clean-datasource-123",
).info("llm_call_start")

# 调用结束后：写日志 + 落库 llm_calls + 打 Prometheus 指标
LLM_TOKENS_TOTAL.labels(model=settings.DEFAULT_LLM_MODEL, type="input").inc(input_tokens)
LLM_COST_USD_TOTAL.labels(model=settings.DEFAULT_LLM_MODEL).inc(total_cost)
LLM_LATENCY.labels(agent="cleaning_agent").observe(duration_seconds)
```

**能力对照**：
- 📊 Token 使用统计 → `llm_tokens_total`
- ⏱️ 延迟分析 → `llm_latency_seconds` + `llm_calls.duration`
- 🔍 调用链路 → `llm_calls`（task_id / attempt / idempotency_key）
- 📈 成本追踪 → `llm_cost_usd_total` + `03-成本管理.md`
- 🐛 错误调试 → Loguru 结构化日志

> 详细设计与阈值见 `09-可观测与成本监控.md`。

## 9. 技术选型理由

### 9.1 为什么选 FastAPI？

✅ **优势**：
- 性能接近 Go/Node.js
- 自动生成文档
- 类型安全
- 异步原生支持
- 学习曲线平缓

❌ **劣势**：
- 生态不如 Django 成熟
- 没有内置 Admin

**对比**：
| 框架 | 性能 | 文档生成 | 异步 | 学习难度 |
|------|------|----------|------|----------|
| FastAPI | ⭐⭐⭐⭐⭐ | ✅ | ✅ | 低 |
| Django | ⭐⭐⭐ | ❌ | 部分 | 中 |
| Flask | ⭐⭐⭐⭐ | ❌ | ❌ | 低 |

---

### 9.2 为什么选 LangChain？

✅ **优势**：
- AI 应用开发的事实标准
- 丰富的集成（50+ LLMs，本项目使用 `langchain-deepseek`）
- 活跃的社区
- 生态配套完善（LangGraph 工作流编排）
- 可观测性可自建（Loguru + Prometheus），不绑定 SaaS

❌ **劣势**：
- 版本迭代快
- API 有时不稳定

**对比**：
| 框架 | 成熟度 | 生态 | 可观测性 | 学习曲线 |
|------|--------|------|----------|----------|
| LangChain | ⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | ⭐⭐⭐⭐⭐ | 中 |
| LlamaIndex | ⭐⭐⭐⭐ | ⭐⭐⭐ | ⭐⭐⭐ | 中 |
| Semantic Kernel | ⭐⭐⭐ | ⭐⭐ | ⭐⭐ | 低 |

---

### 9.3 为什么选 DeepSeek V4.1 Flash？

✅ **优势**：
- 通过官方集成 `langchain-deepseek` 接入，`ChatDeepSeek` 可直接复用 LangChain 生态
- 中文场景表现满足客服对话清洗/评估需求
- 模型 id 配置化（`settings.DEFAULT_LLM_MODEL`），后续可平滑替换

⚠️ **待确认**：
- 具体单价（`P_IN` / `P_OUT`）与上下文长度上限 —— 确认方式：DeepSeek 官方定价页 + 模型文档
- 结构化 JSON 输出的稳定性 —— 确认方式：M2 POC 结构化输出测试（见 `04-开发方案.md` POC 1）

**成本口径**：单价不写死在文档里，统一用 `03-成本管理.md` 的
`月成本 = (月输入 token × P_IN + 月输出 token × P_OUT) / 1_000_000` 计算。

> ⚠️ **已弃用选项（仅历史参考）**：早期版本曾对比 OpenAI GPT / Anthropic Claude 的准确性与定价，
> 相关排序与单价表已删除，不再作为选型依据。如需重新评估，须以各家官方当期定价重新实测。
