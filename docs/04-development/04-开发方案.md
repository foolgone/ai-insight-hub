## 1. 项目概述

### 1.1 项目定位

**AI Insight Hub** 是一个为 AI 客服/对话系统提供数据清洗、质量评估、知识构建和洞察分析的智能数据中台。

### 1.2 核心价值

```
不是审核"人工客服"，而是管理和优化"AI 客服"
通过数据中台能力，让 AI 客服持续进化
```

### 1.3 技术栈

```
前端: React 18 + Vite 5 + TailwindCSS 3
后端: Python 3.11 + FastAPI + SQLAlchemy 2.0
AI:   LangChain + LangGraph + DeepSeek V4.1 Flash（langchain-deepseek）
数据: PostgreSQL 14+ (pgvector) + Redis 7+
部署: Docker + Docker Compose
```

### 1.4 MVP 功能范围

**P0 核心功能**（必须有）：
- ✅ 数据源管理（CRUD）
- ✅ 数据清洗 Agent
- ✅ 质量评估 Agent
- ✅ 知识提取 Agent
- ✅ 基础报告生成

**P1 重要功能**（应该有）：
- ⏸️ 洞察分析 Agent
- ⏸️ 多格式报告导出
- ⏸️ 实时监控 Dashboard

**P2 增强功能**（可以没有）：
- ⏸️ 多用户支持
- ⏸️ API 密钥管理
- ⏸️ Webhook 通知

#### POC 1: LangChain + DeepSeek 集成

**目标**：验证能否成功调用 DeepSeek V4.1 Flash 并使用 LangChain

**任务**：
```python
# backend/tests/poc/test_deepseek_integration.py
import json

from langchain_core.messages import HumanMessage, SystemMessage
from langchain_deepseek import ChatDeepSeek

from app.core.config import settings


def _build_llm() -> ChatDeepSeek:
    """模型 id 统一从配置读取，POC 阶段也不写死"""
    return ChatDeepSeek(
        model=settings.DEFAULT_LLM_MODEL,
        temperature=settings.DEFAULT_TEMPERATURE,
        max_tokens=settings.DEFAULT_MAX_TOKENS,
    )


def test_deepseek_basic():
    """测试基础调用"""
    llm = _build_llm()

    messages = [
        SystemMessage(content="你是一个AI助手"),
        HumanMessage(content="你好"),
    ]

    response = llm.invoke(messages)
    assert response.content
    assert len(response.content) > 0
    print(f"✅ DeepSeek 响应: {response.content}")


def test_deepseek_with_json_output():
    """测试结构化输出"""
    llm = _build_llm()

    prompt = """
    提取以下信息并返回 JSON:
    文本: "用户：iPhone 14 多少钱？客服：5999 元起"

    返回格式: {"user_message": "...", "ai_response": "..."}
    """

    response = llm.invoke(prompt)
    # 验证能够解析 JSON
    data = json.loads(response.content)
    assert "user_message" in data
    assert "ai_response" in data
    print(f"✅ 结构化输出: {data}")
```

**验收标准**：
- ✅ 能成功调用 DeepSeek API
- ✅ 能获取结构化 JSON 输出（解析失败率需记录）
- ✅ Token 使用量可追踪（写入 `llm_calls` 表）

```python
# backend/tests/poc/test_cost_tracking.py
from decimal import Decimal

from langchain_core.callbacks import BaseCallbackHandler
from langchain_deepseek import ChatDeepSeek

from app.core.config import settings


class CostCallback(BaseCallbackHandler):
    """POC 用：验证 token 用量可被捕获并按 P_IN / P_OUT 计费"""

    def __init__(self):
        self.total_cost = Decimal("0")
        self.token_usage = {"input": 0, "output": 0}

    def on_llm_end(self, response, **kwargs):
        llm_output = response.llm_output or {}
        usage = llm_output.get("token_usage") or {}
        input_tokens = usage.get("prompt_tokens", 0)
        output_tokens = usage.get("completion_tokens", 0)

        # 单价来自配置（USD / 百万 token），不在代码里写死
        p_in = Decimal(str(settings.LLM_PRICE_INPUT_PER_MTOK))
        p_out = Decimal(str(settings.LLM_PRICE_OUTPUT_PER_MTOK))

        input_cost = Decimal(input_tokens) / Decimal(1_000_000) * p_in
        output_cost = Decimal(output_tokens) / Decimal(1_000_000) * p_out

        self.token_usage["input"] += input_tokens
        self.token_usage["output"] += output_tokens
        self.total_cost += (input_cost + output_cost)


def test_cost_tracking():
    """测试成本追踪"""
    callback = CostCallback()
    llm = ChatDeepSeek(model=settings.DEFAULT_LLM_MODEL)
    
    llm.invoke(
        "用 50 字介绍 Python",
        config={"callbacks": [callback]}
    )
    
    assert callback.total_cost >= 0
    assert callback.token_usage["input"] > 0
    assert callback.token_usage["output"] > 0
    
    print(f"✅ Token 使用: {callback.token_usage}")
    print(f"✅ 成本: ${callback.total_cost:.8f}")
```

```python
from pathlib import Path

from app.agents.base import BaseAgent
from app.core.config import settings

class DataCleaningAgent(BaseAgent):
    """数据清洗 Agent"""
    
    PROMPT_DIR = Path(__file__).parent.parent / "prompts" / "cleaning"
    
    def __init__(self):
        super().__init__(
            model=settings.DEFAULT_LLM_MODEL,      # 模型 id 从配置读取
            temperature=settings.DEFAULT_TEMPERATURE,
        )
    
    def _load_prompt(self):
        """加载 Prompt"""
        return (self.PROMPT_DIR / "system.txt").read_text(encoding="utf-8")
    
    async def clean(self, raw_data: str) -> dict:
        """清洗数据"""
        prompt = self._load_prompt()
        
        messages = [
            {"role": "system", "content": prompt},
            {"role": "user", "content": f"清洗以下数据：\n{raw_data}"}
        ]
        
        response = await self.llm.ainvoke(messages)
        
        # 解析输出
        import json
        return json.loads(response.content)
```

```python
# app/agents/evaluation_agent.py
class QualityEvaluationAgent(BaseAgent):
    """质量评估 Agent"""
    
    DIMENSIONS = ["accuracy", "completeness", "hallucination", "tone", "compliance"]
    
    async def evaluate(self, dialogue: dict) -> dict:
        """评估质量"""
        results = {}
        
        for dimension in self.DIMENSIONS:
            prompt = self._load_dimension_prompt(dimension)
            score = await self._evaluate_dimension(dialogue, prompt)
            results[dimension] = score
        
        # 计算加权总分
        overall_score = self._calculate_overall(results)
        
        return {
            "overall_score": overall_score,
            "dimensions": results,
            "needs_review": overall_score < 70
        }
```

**验收标准**：
- ✅ 能评估 5 个维度（accuracy / completeness / tone / compliance / hallucination）
- ✅ 准确率 > 85%（需人工标注集验证）

```python
# app/agents/extraction_agent.py
from sentence_transformers import SentenceTransformer

class KnowledgeExtractionAgent(BaseAgent):
    """知识提取 Agent"""
    
    def __init__(self):
        super().__init__()
        self.embedding_model = SentenceTransformer(
            'paraphrase-multilingual-MiniLM-L12-v2'
        )
    
    async def extract(self, dialogues: list) -> dict:
        """提取知识"""
        # 1. 提取问题
        questions = await self._extract_questions(dialogues)
        
        # 2. 聚类
        clusters = self._cluster_similar(questions)
        
        # 3. 提取标准答案
        faqs = []
        for cluster in clusters:
            answer = await self._extract_answer(cluster)
            embedding = self.embedding_model.encode(cluster[0]['question'])
            
            faqs.append({
                "question": cluster[0]['question'],
                "answer": answer,
                "frequency": len(cluster),
                "embedding": embedding.tolist()
            })
        
        return {"faqs": faqs}
```
