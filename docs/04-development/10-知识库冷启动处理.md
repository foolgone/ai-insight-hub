## 1. 问题描述

### 当前问题

新数据源刚创建时：

❌ 知识库为空，无法进行向量搜索  
❌ 用户不知道为什么搜索不到结果  
❌ 没有引导用户完成知识提取  
❌ 用户体验差  

### 影响

- 用户创建数据源后立即使用知识检索功能会报错或返回空结果
- 用户不理解为什么没有结果
- 可能误以为系统有问题

---

## 2. 解决方案

### 2.1 三级方案

```
方案 1: 种子知识库（通用 FAQ）
方案 2: 智能引导（提示用户完成步骤）
方案 3: 优雅降级（返回空结果 + 友好提示）
```

class OnboardingService:
    """新手引导服务"""
    
    async def get_next_step(self, datasource_id: str) -> dict:
        """获取下一步操作建议"""
        
        datasource = await self.db.fetch_one(
            "SELECT * FROM data_sources WHERE id = :id",

        )
        
        status = datasource["workflow_status"]
        
        if status == "created":
            return {
                "step": 1,
                "title": "上传数据",
                "description": "请上传客服对话数据（CSV/JSON/Excel）",
                "action": {
                    "type": "upload",

                },
                "can_skip": False
            }
        
        elif status == "data_uploaded":
            return {
                "step": 2,
                "title": "数据清洗",
                "description": "对上传的数据进行清洗和标准化",
                "action": {
                    "type": "clean",
                    "url": f"/datasources/{datasource_id}/clean"
                },
                "can_skip": False,
                "auto_trigger": True  # 可自动触发
            }
        
        elif status == "data_cleaned":
            return {
                "step": 3,
                "title": "构建知识库",
                "description": "从对话中提取 FAQ 知识库",
                "action": {

                },
                "can_skip": False,

            return {
                "step": 4,
                "title": "完成",

## 3. 实现代码

### 3.1 初始化种子知识库

```python
# app/services/datasource_service.py
class DataSourceService:
    """数据源服务"""
    
    async def create(self, data: DataSourceCreate, user_id: str) -> DataSource:
        """创建数据源"""
        
        # 1. 创建数据源
        datasource = await self.db.execute(
            """
            INSERT INTO data_sources (
                name, type, description, user_id, workflow_status
            ) VALUES (
                :name, :type, :description, :user_id, 'created'
            ) RETURNING *
            """,
            {
                "name": data.name,
                "type": data.type,
                "description": data.description,
                "user_id": user_id
            }
        )
        
        # 2. 初始化种子知识库（可选）
        if data.config.get("use_seed_knowledge", True):
            await self._init_seed_knowledge(datasource["id"])
        
        return datasource
    
    async def _init_seed_knowledge(self, datasource_id: str):
        """初始化种子知识库"""
        from app.data.seed_knowledge import get_seed_faqs
        from sentence_transformers import SentenceTransformer
        
        # 1. 获取种子 FAQ
        seed_faqs = get_seed_faqs()
        
        # 2. 生成向量
        model = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
        
        # 3. 批量插入
        for faq in seed_faqs:
            embedding = model.encode(faq["question"]).tolist()
            
            await self.db.execute(
                """
                INSERT INTO knowledge_base (
                    datasource_id, question, answer, category,
                    embedding, frequency, is_seed
                ) VALUES (
                    :datasource_id, :question, :answer, :category,
                    :embedding, 0, TRUE
                )
                """,
                {
                    "datasource_id": datasource_id,
                    "question": faq["question"],
                    "answer": faq["answer"],
                    "category": faq["category"],
                    "embedding": embedding
                }
            )
        
        logger.info(f"Initialized {len(seed_faqs)} seed FAQs for datasource {datasource_id}")
```

**注意**：种子知识库标记为 `is_seed=True`，可以在后续被用户数据替换。

### 3.3 知识提取时替换种子知识

```python
# app/agents/extraction_agent.py
class KnowledgeExtractionAgent(BaseAgent):
    """知识提取 Agent"""
    
    async def extract(self, datasource_id: str, dialogues: list) -> dict:
        """提取知识"""
        
        # 1. 提取 FAQ
        faqs = await self._extract_faqs(dialogues)
        
        # 2. 清理种子知识库
        await self.db.execute(
            "DELETE FROM knowledge_base WHERE datasource_id = :id AND is_seed = TRUE",
            {"id": datasource_id}
        )
        
        logger.info(f"Cleared seed knowledge for datasource {datasource_id}")
        
        # 3. 插入真实知识
        for faq in faqs:
            embedding = self.embedding_model.encode(faq["question"]).tolist()
            
            await self.db.execute(
                """
                INSERT INTO knowledge_base (
                    datasource_id, question, answer, category,
                    embedding, frequency, is_seed
                ) VALUES (
                    :datasource_id, :question, :answer, :category,
                    :embedding, :frequency, FALSE
                )
                """,
                {
                    "datasource_id": datasource_id,
                    "question": faq["question"],
                    "answer": faq["answer"],
                    "category": faq["category"],
                    "embedding": embedding,
                    "frequency": faq["frequency"]
                }
            )
        
        # 4. 更新数据源状态
        await self.db.execute(
            """
            UPDATE data_sources
            SET workflow_status = 'knowledge_built'
            WHERE id = :id
            """,
            {"id": datasource_id}
        )
        
        return {"total_faqs": len(faqs)}
```

// src/components/onboarding/DataSourceOnboarding.jsx
import { useEffect, useState } from 'react';
import { useOnboardingSteps } from '@/hooks/useOnboarding';

export function DataSourceOnboarding({ datasourceId }) {
  const { steps, currentStep, isComplete } = useOnboardingSteps(datasourceId);
  
  if (isComplete) {
    return null; // 完成后不显示
  }
  
  return (
    <div className="onboarding-card">
      <h3>设置向导</h3>
      
      <div className="steps">
        {steps.map((step, index) => (
          <div
            key={index}
            className={`step ${step.completed ? 'completed' : ''} ${
              index === currentStep ? 'active' : ''
            }`}
          >
            <div className="step-number">{index + 1}</div>
            <div className="step-content">
              <h4>{step.title}</h4>
              <p>{step.description}</p>
              
              {index === currentStep && step.action && (
                <button
                  onClick={() => handleStepAction(step.action)}
                  className="btn-primary"
                >
                  {step.action.type === 'upload' && '上传数据'}
                  {step.action.type === 'clean' && '开始清洗'}
                  {step.action.type === 'extract' && '构建知识库'}
                </button>
              )}
              
              {step.estimated_time && (
                <span className="estimated-time">
                  预计耗时：{step.estimated_time}
                </span>
              )}
            </div>
          </div>
        ))}
      </div>
      
      {currentStep < steps.length && (
        <div className="progress-bar">
          <div
            className="progress-fill"
            style={{ width: `${(currentStep / steps.length) * 100}%` }}
          />
        </div>
      )}
    </div>
  );
}
