## 1. 问题描述

### 当前问题

新数据源刚创建时：

❌ 知识库为空，无法进行向量搜索  
❌ 用户不知道为什么搜索不到结果  
❌ 没有引导用户完成知识提取  
❌ 用户体验差  

### 影响

- 用户创建数据源后立即使用知识检索功能会报错或返回空结果
- 用户不理解为什么没有结果
- 可能误以为系统有问题

---

## 2. 解决方案

### 2.1 三级方案

```
方案 1: 种子知识库（通用 FAQ）
方案 2: 智能引导（提示用户完成步骤）
方案 3: 优雅降级（返回空结果 + 友好提示）
```

### 2.1.1 冷启动判定阈值

「知识库为空」不足以覆盖真实场景：只有 3 条 FAQ 时，检索体验和空库差别不大。因此用**有效 FAQ 条数**判定：

| FAQ 条数（`status='active'` 且非种子） | 状态 | 行为 |
|--------------------------------------|------|------|
| `0` | `empty` | 返回空结果 + 引导 + 提示可用种子知识库 |
| `1 – 49` | `cold_start` | 正常检索，但顶部提示「知识库较稀疏，结果可能不完整」，并推荐人工补录 |
| `>= 50` | `ready` | 正常检索，不提示 |

```python
# app/services/knowledge_status.py
COLD_START_THRESHOLD = 50  # 低于 50 条视为冷启动


def resolve_knowledge_status(active_faq_count: int) -> str:
    """根据有效 FAQ 条数返回知识库状态。"""
    if active_faq_count <= 0:
        return "empty"
    if active_faq_count < COLD_START_THRESHOLD:
        return "cold_start"
    return "ready"
```

> 50 条是**经验阈值（待验证）**：上线后用「检索命中率 / 空结果率」回测，再决定是否调整；调整时只改 `COLD_START_THRESHOLD` 一处。

### 2.1.2 人工补录流程

冷启动阶段最有效的补充不是种子 FAQ，而是**运营/客服手工补录高频问题**：

1. **入口**：知识库页面「新增 FAQ」+ 批量导入（CSV：`question,answer,category`）；
2. **校验**：`question` 去重（`pg_trgm` 相似度 > 0.9 视为重复）、`answer` 非空、长度上限 2000 字；
3. **标记**：手工补录写 `is_seed = FALSE`、`source = 'manual'`（`source` 列已在 `03-数据库设计.md` §3.6 定义）；
4. **立即生效**：补录后同步生成 embedding（384 维）并写入 `knowledge_base`，无需等全量知识提取；
5. **可追溯**：`audit_logs` 记录操作人、时间、变更前后内容；
6. **目标**：冷启动数据源在 3 个工作日内补到 ≥ 50 条，之后由知识提取任务增量覆盖。

### 2.4 优雅降级

#### API 响应

```python
# app/api/v1/knowledge.py
from fastapi import APIRouter, Depends, Query
from loguru import logger

from app.core.db import get_db
from app.services.knowledge_status import COLD_START_THRESHOLD, resolve_knowledge_status
from app.services.onboarding_service import OnboardingService

router = APIRouter()


@router.get("/datasources/{datasource_id}/knowledge/search")
async def search_knowledge(
    datasource_id: str,
    q: str = Query(..., min_length=1, max_length=200),
    limit: int = Query(10, ge=1, le=50),
    db=Depends(get_db),
):
    """搜索知识库"""

    # 1. 统计有效 FAQ 条数（排除种子知识）
    count = await db.fetch_val(
        """
        SELECT COUNT(*)
        FROM knowledge_base
        WHERE datasource_id = :id
          AND status = 'active'
          AND is_seed = FALSE
        """,
        {"id": datasource_id},
    )
    count = int(count or 0)
    knowledge_status = resolve_knowledge_status(count)

    # 2. 空库：返回友好提示
    if knowledge_status == "empty":
        next_step = await OnboardingService(db).get_next_step(datasource_id)
        logger.bind(datasource_id=datasource_id, status=knowledge_status).info(
            "knowledge_search_empty"
        )
        return {
            "results": [],
            "total": 0,
            "status": "empty",
            "message": "知识库还未构建",
            "suggestion": {
                "title": "如何构建知识库？",
                "steps": [
                    "1. 上传客服对话数据",
                    "2. 运行数据清洗",
                    "3. 提取知识库",
                ],
                "next_step": next_step,
                "can_use_seed": True,  # 可以使用种子知识库
            },
        }

    # 3. 执行搜索（种子知识也参与，但排在真实知识之后）
    results = await search_in_knowledge_base(datasource_id, q, limit)

    payload = {
        "results": results,
        "total": len(results),
        "status": knowledge_status,
    }

    # 4. 冷启动：正常返回，但附加提示
    if knowledge_status == "cold_start":
        payload["message"] = (
            f"当前知识库仅 {count} 条（少于 {COLD_START_THRESHOLD} 条），"
            "结果可能不完整，建议补充高频问题。"
        )
        payload["suggestion"] = {
            "title": "补充知识库",
            "steps": ["手工新增 FAQ", "批量导入 CSV", "继续从对话中提取"],
        }

    return payload
```

## 3. 实现代码

### 3.1 初始化种子知识库

```python
# app/services/datasource_service.py
from loguru import logger
from sentence_transformers import SentenceTransformer

from app.data.seed_knowledge import get_seed_faqs
from app.schemas.datasource import DataSourceCreate


class DataSourceService:
    """数据源服务"""

    def __init__(self, db, embedding_model: SentenceTransformer):
        self.db = db
        self.embedding_model = embedding_model

    async def create(self, data: DataSourceCreate, user_id: str) -> dict:
        """创建数据源"""

        # 1. 创建数据源
        datasource = await self.db.fetch_one(
            """
            INSERT INTO data_sources (
                name, type, description, user_id, workflow_status
            ) VALUES (
                :name, :type, :description, :user_id, 'created'
            ) RETURNING *
            """,
            {
                "name": data.name,
                "type": data.type,
                "description": data.description,
                "user_id": user_id,
            },
        )

        # 2. 初始化种子知识库（可选，默认开启）
        if (data.config or {}).get("use_seed_knowledge", True):
            await self._init_seed_knowledge(datasource["id"])

        return datasource

    async def _init_seed_knowledge(self, datasource_id: str) -> None:
        """初始化种子知识库（幂等：重复调用不会重复插入）"""

        seed_faqs = get_seed_faqs()

        for faq in seed_faqs:
            embedding = self.embedding_model.encode(faq["question"]).tolist()

            await self.db.execute(
                """
                INSERT INTO knowledge_base (
                    datasource_id, question, answer, category,
                    embedding, frequency, is_seed
                )
                SELECT
                    :datasource_id, :question, :answer, :category,
                    :embedding, 0, TRUE
                WHERE NOT EXISTS (
                    SELECT 1 FROM knowledge_base
                    WHERE datasource_id = :datasource_id
                      AND question = :question
                      AND is_seed = TRUE
                )
                """,
                {
                    "datasource_id": datasource_id,
                    "question": faq["question"],
                    "answer": faq["answer"],
                    "category": faq["category"],
                    "embedding": embedding,
                },
            )

        logger.bind(datasource_id=datasource_id, count=len(seed_faqs)).info(
            "seed_knowledge_initialized"
        )
```

**注意**：种子知识库标记为 `is_seed=True`，可以在后续被用户数据替换。

### 3.2 数据库 Schema 更新

以下 DDL 是本文方案新增的字段/函数，**已同步到 `docs/03-architecture/03-数据库设计.md`**（`data_sources.workflow_status` 见 §3.2，`knowledge_base.is_seed` / `source` 及索引见 §3.6）。

```sql
-- 1. 知识库：种子标记 + 来源（manual / extracted / seed）
ALTER TABLE knowledge_base ADD COLUMN IF NOT EXISTS is_seed BOOLEAN DEFAULT FALSE;
ALTER TABLE knowledge_base ADD COLUMN IF NOT EXISTS source VARCHAR(20) DEFAULT 'extracted'
    CHECK (source IN ('seed', 'manual', 'extracted'));

-- 2. 数据源：工作流状态（引导流程用）
ALTER TABLE data_sources ADD COLUMN IF NOT EXISTS workflow_status VARCHAR(50) DEFAULT 'created'
    CHECK (workflow_status IN (
        'created', 'data_uploaded', 'data_cleaned', 'knowledge_built', 'ready'
    ));

-- 3. 索引
CREATE INDEX IF NOT EXISTS idx_kb_is_seed
    ON knowledge_base(datasource_id, is_seed);
CREATE INDEX IF NOT EXISTS idx_kb_source
    ON knowledge_base(datasource_id, source);

-- 4. 清理种子知识库的函数
CREATE OR REPLACE FUNCTION clear_seed_knowledge(p_datasource_id UUID)
RETURNS void AS $$
BEGIN
    DELETE FROM knowledge_base
    WHERE datasource_id = p_datasource_id AND is_seed = TRUE;
END;
$$ LANGUAGE plpgsql;
```

> ⚠️ `data_sources.status` 已有值域 `active/archived/processing/error`，与 `workflow_status` 是**两个维度**：前者表示数据源生命周期，后者表示引导进度，不要合并。

### 3.3 知识提取时替换种子知识

```python
# app/agents/extraction_agent.py
from loguru import logger

from app.agents.base import BaseAgent


class KnowledgeExtractionAgent(BaseAgent):
    """知识提取 Agent"""

    async def extract(self, datasource_id: str, dialogues: list) -> dict:
        """提取知识"""

        # 1. 提取 FAQ
        faqs = await self._extract_faqs(dialogues)

        # 2. 清理种子知识库（只删 is_seed=TRUE，手工补录保留）
        await self.db.execute(
            "DELETE FROM knowledge_base WHERE datasource_id = :id AND is_seed = TRUE",
            {"id": datasource_id},
        )
        logger.bind(datasource_id=datasource_id).info("seed_knowledge_cleared")

        # 3. 插入真实知识（按 question 去重，避免覆盖手工补录）
        inserted = 0
        for faq in faqs:
            embedding = self.embedding_model.encode(faq["question"]).tolist()

            await self.db.execute(
                """
                INSERT INTO knowledge_base (
                    datasource_id, question, answer, category,
                    embedding, frequency, is_seed, source
                ) VALUES (
                    :datasource_id, :question, :answer, :category,
                    :embedding, :frequency, FALSE, 'extracted'
                )
                ON CONFLICT DO NOTHING
                """,
                {
                    "datasource_id": datasource_id,
                    "question": faq["question"],
                    "answer": faq["answer"],
                    "category": faq["category"],
                    "embedding": embedding,
                    "frequency": faq["frequency"],
                },
            )
            inserted += 1

        # 4. 更新数据源状态
        await self.db.execute(
            """
            UPDATE data_sources
            SET workflow_status = 'knowledge_built',
                updated_at = NOW()
            WHERE id = :id
            """,
            {"id": datasource_id},
        )

        return {"total_faqs": inserted}
```

> 去重依赖 `(datasource_id, question)` 唯一约束；若尚未建该约束，先用 `WHERE NOT EXISTS` 兜底，或在迁移里补 `CREATE UNIQUE INDEX ... ON knowledge_base(datasource_id, question)`。

```jsx
import { useNavigate } from 'react-router-dom';

import { useOnboardingSteps } from '@/hooks/useOnboarding';

            <div className="step-number">{index + 1}</div>

              <h4>{step.title}</h4>

                <button
                  onClick={() => handleStepAction(step.action)}
                  className="btn-primary"
                >
                  {step.action.type === 'upload' && '上传数据'}
                  {step.action.type === 'clean' && '开始清洗'}

                  预计耗时：{step.estimated_time}
                </span>
              )}
            </div>
          </div>
        ))}
      </div>
      
      {currentStep < steps.length && (
        <div className="progress-bar">
          <div
            className="progress-fill"
            style={{ width: `${(currentStep / steps.length) * 100}%` }}
          />
        </div>
      )}
    </div>
  );
}
```

```jsx
      <p className="status-text">{task?.currentStep}</p>

      <div className="estimated-time">
        <HiOutlineClock className="icon" />
        <span>预计还需 {task?.estimatedRemaining ?? '-'} 分钟</span>
      </div>

      {task?.status === 'completed' && (
        <div className="success-message">
          <HiOutlineCheckCircle className="icon" />
          <span>知识库构建完成！共提取 {task?.result?.totalFaqs ?? 0} 个 FAQ</span>
        </div>
      )}
    </div>
  );
}
```

> 图标统一用 `react-icons`（`frontend/package.json` 已安装），不要额外引入 `@heroicons/react`。
